function on_transfer_attempt(ctx, request)
    propose_route!(ctx, request,
        choose_route(ctx, request))
    hook_before_transfer!(ctx, request)    destination = requested_route(ctx, request)
    outcome = try_transfer!(ctx,
        request, destination)

    if is_waiting(outcome)
        register_wait!(ctx, request, outcome)
    end
end
